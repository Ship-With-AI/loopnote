```ts
export function fileTypeMarkFor(displayPath: string | undefined): Readonly<{ text: string; tone: string }> {
  const extension = displayPath?.match(/\.([^./]+)$/)?.[1]?.toLowerCase();
  return extension !== undefined && extension in FILE_TYPE_MARKS
    ? FILE_TYPE_MARKS[extension]
    : FALLBACK_FILE_TYPE_MARK;
}
```

This is the same regex/lookup `FileRow.vue`'s own `fileTypeMark` computed (lines 94-100) already uses, just taking the resolved display-path string directly as an argument instead of reading `props.leaf.file.newPath?.display ?? props.leaf.file.oldPath?.display` itself — the caller resolves which path to pass (so `file-tree.ts` stays free of any `FileTreeLeaf`-shaped dependency it doesn't already have).
